<script>
  window.addEventListener('load', () => { setTimeout(() => { document.getElementById('splash-screen').classList.add('fade-out'); }, 2500); });

  let db = JSON.parse(localStorage.getItem('salca_full_db')) || [];
  function saveDB() { localStorage.setItem('salca_full_db', JSON.stringify(db)); }

  const formatLongDate = (dateStr) => {
    if(!dateStr) return 'N/A';
    const date = new Date(dateStr);
    return date.toLocaleDateString('es-ES', { day: 'numeric', month: 'long', year: 'numeric' });
  };

  function updateTable() {
    const tbody = document.querySelector('#membersTable tbody');
    tbody.innerHTML = '';
    db.forEach((m, i) => {
      tbody.insertAdjacentHTML('beforeend', `
        <tr>
          <td style='font-weight:600;'>${m.fullName}</td>
          <td style='color:#888;'>${m.idNumber}</td>
          <td class='actions-cell' style='display:flex; gap:8px;'>
            <button class='btn btn-main' onclick='openPayments(${i})'>PAGOS</button>
            <button class='btn btn-outline' onclick='openProfile(${i})'>FICHA</button>
            <button class='btn btn-danger' onclick='deleteMember(${i})'>✕</button>
          </td>
        </tr>`);
    });
  }

  window.deleteMember = (i) => { if(confirm(`¿Eliminar a ${db[i].fullName}?`)) { db.splice(i, 1); saveDB(); updateTable(); } };

  window.addNewMonth = (i) => {
    const hoy = new Date();
    let mesSig = new Date(hoy.getFullYear(), hoy.getMonth() + 1, 5);
    db[i].months.push({ mes: 'Mensualidad', estado: 'Pendiente', fechaVence: mesSig.toISOString() });
    saveDB(); openPayments(i);
  };

  window.openPayments = (i) => {
    const m = db[i];
    let rows = m.months.map((mon, mi) => `<div class='payment-row'><div><b>${mon.mes}</b><br><small style='color:#888;'>Vence: ${formatLongDate(mon.fechaVence)}</small></div><div>${mon.estado === 'Pagado' ? `<span style='color:#34c759; margin-right:10px;'>PAGADO</span><button class='btn' onclick='printReceipt(${i},${mi})'>RECIBO</button>` : `<button class='btn btn-main' onclick='payMonth(${i},${mi})'>PAGAR</button>`}</div></div>`).join('');
    document.getElementById('modalBody').innerHTML = `<h2 style='font-family:Cinzel; margin:0;'>Pagos: ${m.fullName}</h2><button class='btn btn-main' style='width:100%; margin:20px 0;' onclick='addNewMonth(${i})'>+ AGREGAR MES</button><div style='max-height:300px; overflow-y:auto;'>${rows || 'Sin registros.'}</div><br><button class='btn' style='background:#222; color:#fff;' onclick='closeModal()'>CERRAR</button>`;
    document.getElementById('modal').style.display = 'flex';
  };

  window.openProfile = (i) => {
    const m = db[i];
    document.getElementById('modalBody').innerHTML = `
      <h2 style='font-family:Cinzel; margin:0;'>Ficha Técnica</h2>
      <div style='margin-top:20px; max-height: 450px; overflow-y: auto;'>
        <div class='grid-2'><div><label>Teléfono:</label><input type='text' id='tel' value='${m.tel||''}'></div><div><label>Nacimiento:</label><input type='date' id='birth' value='${m.birth||''}'></div></div>
        <div class='grid-2'><div><label>EPS:</label><input type='text' id='eps' value='${m.eps||''}'></div><div><label>Grupo:</label><select id='groupType'><option ${m.groupType==='Baile Social'?'selected':''}>Baile Social</option><option ${m.groupType==='Grupo Competitivo'?'selected':''}>Grupo Competitivo</option></select></div></div>
        <label>Responsable:</label><input type='text' id='respName' value='${m.respName||''}'>
        <label>Tel. Responsable:</label><input type='text' id='respTel' value='${m.respTel||''}'>
        <label>Nivel:</label><select id='level'><option ${m.level==='Básico'?'selected':''}>Básico</option><option ${m.level==='Intermedio'?'selected':''}>Intermedio</option><option ${m.level==='Avanzado'?'selected':''}>Avanzado</option></select>
        <label>Salud/Observaciones:</label><textarea id='health' rows='3'>${m.health||''}</textarea>
      </div>
      <button class='btn btn-main' style='width:100%; margin-top:20px;' onclick='saveProfile(${i})'>GUARDAR</button>
      <button class='btn' style='background:#222; color:#fff; width:100%; margin-top:10px;' onclick='closeModal()'>CANCELAR</button>`;
    document.getElementById('modal').style.display = 'flex';
  };

  window.saveProfile = (i) => {
    db[i].tel = document.getElementById('tel').value; db[i].birth = document.getElementById('birth').value; db[i].eps = document.getElementById('eps').value; db[i].groupType = document.getElementById('groupType').value;
    db[i].respName = document.getElementById('respName').value; db[i].respTel = document.getElementById('respTel').value; db[i].health = document.getElementById('health').value; db[i].level = document.getElementById('level').value;
    saveDB(); closeModal(); alert('Actualizado');
  };

  window.payMonth = (mi, moni) => { db[mi].months[moni].estado = 'Pagado'; db[mi].months[moni].fechaPago = new Date().toISOString(); saveDB(); openPayments(mi); };

  window.printReceipt = (mi, moni) => {
    const m = db[mi]; const p = m.months[moni];
    const win = window.open('', '', 'width=600,height=550');
    win.document.write("<html><head><style>body { font-family: Montserrat; text-align: center; padding: 20px; color: #333; } .receipt-box { border: 2px solid #000; padding: 25px; min-height: 400px; position: relative; } .title { font-family: Cinzel; font-size: 2.2em; font-weight: bold; margin-bottom: 5px; } .subtitle { letter-spacing: 2px; font-size: 0.8em; margin-bottom: 20px; } .details { text-align: left; margin-top: 30px; line-height: 1.8; font-size: 1.1em; border-top: 1px dashed #ccc; padding-top: 15px; } .bottom-container { display: flex; justify-content: space-between; flex-direction: row-reverse; align-items: flex-end; margin-top: 40px; } .signature-box { width: 160px; height: 60px; border: 1px solid #000; position: relative; } .signature-label { position: absolute; bottom: -20px; width: 100%; font-size: 0.7em; font-weight: bold; text-align: center; } .stamp { border: 2px solid #000; padding: 5px 15px; font-weight: bold; text-transform: uppercase; font-size: 0.8em; }</style></head><body><div class='receipt-box'><div class='title'>SALCA</div><div class='subtitle'>FUNDACIÓN Y ESCUELA DE BAILE</div><div class='details'><b>ALUMNO:</b> " + m.fullName.toUpperCase() + "<br><b>CONCEPTO:</b> PAGO DE MENSUALIDAD<br><b>FECHA DE PAGO:</b> " + formatLongDate(p.fechaPago) + "<br><b>VIGENCIA HASTA:</b> " + formatLongDate(p.fechaVence) + "</div><div class='bottom-container'><div class='signature-box'><div class='signature-label'>FIRMA O SELLO</div></div><div class='stamp'>Comprobante Oficial</div></div></div></body></html>");
    win.document.close(); setTimeout(() => { win.print(); win.close(); }, 800);
  };

  function saveMember() {
    const n = document.getElementById('fullName').value; if(!n) return;
    db.push({ fullName: n, idNumber: document.getElementById('idNumber').value, months: [], health: '', tel: '', level: 'Básico', groupType: 'Baile Social' });
    saveDB(); updateTable(); document.getElementById('fullName').value = ''; document.getElementById('idNumber').value = '';
  }
  function closeModal() { document.getElementById('modal').style.display = 'none'; }
  updateTable();
</script>
